cmake_minimum_required(VERSION 3.22)
if(POLICY CMP0135)
  cmake_policy(SET CMP0135 NEW)
endif()
project(ninfer_portable_gguf LANGUAGES C CXX)
# This is a separate llama.cpp application, NOT a port of NInfer's CUDA engine.
option(NINFER_PORTABLE_VULKAN "Enable the llama.cpp Vulkan backend" OFF)
include(FetchContent)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "Static portable dependencies")
set(GGML_NATIVE OFF CACHE BOOL "Do not inherit the build machine's ISA" FORCE)
set(GGML_AVX ON CACHE BOOL "Zen 3+ AVX" FORCE)
set(GGML_AVX2 ON CACHE BOOL "Zen 3+ AVX2" FORCE)
set(GGML_FMA ON CACHE BOOL "Zen 3+ FMA" FORCE)
set(GGML_F16C ON CACHE BOOL "Zen 3+ F16C" FORCE)
foreach(feature AVX512 AVX512_VBMI AVX512_VNNI AVX512_BF16 AVX_VNNI AVX_VNNI_INT8 BMI2)
  set(GGML_${feature} OFF CACHE BOOL "Conservative Zen 3+ target" FORCE)
endforeach()
set(GGML_CUDA OFF CACHE BOOL "No NVIDIA dependency" FORCE)
set(GGML_HIP OFF CACHE BOOL "No unqualified ROCm path" FORCE)
set(GGML_VULKAN ${NINFER_PORTABLE_VULKAN} CACHE BOOL "Opt-in Vulkan" FORCE)
set(LLAMA_BUILD_COMMON ON CACHE BOOL "Build tools support" FORCE)
set(LLAMA_BUILD_TOOLS ON CACHE BOOL "Build CLI/server/benchmark targets" FORCE)
set(LLAMA_BUILD_SERVER ON CACHE BOOL "Build server" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "No examples" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "No upstream test suite" FORCE)
set(LLAMA_BUILD_APP OFF CACHE BOOL "No unified app" FORCE)
set(LLAMA_BUILD_UI OFF CACHE BOOL "No UI dependency download" FORCE)
set(LLAMA_OPENSSL OFF CACHE BOOL "Local models only; no HTTPS dependency" FORCE)
FetchContent_Declare(llama_cpp
  URL https://codeload.github.com/ggml-org/llama.cpp/tar.gz/900f01ba09ac5735e16a634a2c37e000639c936a
  URL_HASH SHA256=a2feb9d49bb76dd941ce763714612dd8b2fbf9c1e3209bea621f233fc4aa9f65)
FetchContent_MakeAvailable(llama_cpp)
